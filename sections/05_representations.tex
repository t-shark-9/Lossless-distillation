\section{Beyond output matching: intermediate representations}\label{sec:repr}
% ===========================================================================

\subsection{Formalisation}

Let $h_\TT(s)\in\R^{d_\TT}$ be a teacher representation at some layer and $h_\TS(s)\in\R^{d_\TS}$ a student
representation, both centred under a prefix law $\nu$, with covariances $\Sigma_{\TT\TT}$, $\Sigma_{\TS\TS}\succ0$ and
cross-covariance $\Sigma_{\TT\TS}$. Representation matching adds a constraint to the output objective $\cJ$:
\[
\min_\theta\ \cJ(\theta)\quad\text{s.t.}\quad\cR(\theta)\le\eta,
\qquad\text{or its Lagrangian}\quad\cJ(\theta)+\gamma\,\cR(\theta).
\]
Two standard choices are a learned linear map, as in FitNets \citep{romero2015fitnets},
\[
\cR_{\mathrm{LS}}(\theta):=\min_{A\in\R^{d_\TT\times d_\TS}}\E_\nu\|Ah_\TS(s)-h_\TT(s)\|^2,
\]
and linear CKA \citep{kornblith2019similarity},
\[
\cR_{\mathrm{CKA}}(\theta):=1-\frac{\|\Sigma_{\TS\TT}\|_F^2}{\|\Sigma_{\TS\TS}\|_F\,\|\Sigma_{\TT\TT}\|_F}.
\]
Relation-based variants match attention maps or value relations \citep{zagoruyko2017paying,jiao2020tinybert,wang2020minilm,sun2020mobilebert}.
Contrastive variants maximise a mutual-information bound \citep{tian2020contrastive}.

\begin{proposition}[The learned-map penalty is CCA; CKA constrains geometry]\label{prop:cca}
\Pv\Ck
$\cR_{\mathrm{LS}}=\tr\Sigma_{\TT\TT}-\tr(\Sigma_{\TT\TS}\Sigma_{\TS\TS}^{-1}\Sigma_{\TS\TT})$. If the teacher features are whitened
($\Sigma_{\TT\TT}=I$), then $\cR_{\mathrm{LS}}=d_\TT-\sum_i\rho_i^2$, where the $\rho_i$ are the canonical correlations.
$\cR_{\mathrm{LS}}$ is invariant under $h_\TS\mapsto Gh_\TS$ for every invertible $G$, so it constrains only the
\emph{linear information content} of $h_\TS$. $\cR_{\mathrm{CKA}}$ is invariant only under orthogonal maps and
isotropic scaling, so it also constrains the \emph{geometry} (spectrum) of $h_\TS$. For example, with
$d_\TS=d_\TT$ and $h_\TS=\Sigma_{\TT\TT}^{-1/2}h_\TT$: $\cR_{\mathrm{LS}}=0$, but
$1-\cR_{\mathrm{CKA}}=\tr\Sigma_{\TT\TT}/(\sqrt{d_\TT}\,\|\Sigma_{\TT\TT}\|_F)<1$ unless $\Sigma_{\TT\TT}\propto I$.
\end{proposition}
\begin{proof}
Normal equations give $A^\star=\Sigma_{\TT\TS}\Sigma_{\TS\TS}^{-1}$. With $\Sigma_{\TT\TT}=I$,
$\tr(\Sigma_{\TT\TS}\Sigma_{\TS\TS}^{-1}\Sigma_{\TS\TT})=\|\Sigma_{\TS\TS}^{-1/2}\Sigma_{\TS\TT}\|_F^2=\sum_i\rho_i^2$. In the
example, $\Sigma_{\TS\TS}=I$ and $\Sigma_{\TS\TT}=\Sigma_{\TT\TT}^{1/2}$. The CKA inequality is Cauchy--Schwarz on the
eigenvalues.
\end{proof}

\subsection{When representation matching is redundant}

\begin{proposition}[Last-layer matching is implied by high-temperature logit matching]\label{prop:lastlayer}
\Pa{\cref{as:linear}} If $CW_\TT$ has full column rank $d_\TT$ (which requires $V-1\ge d_\TT$, true for
language models), then
\[
\cR_{\mathrm{LS}}\le\E\big\|(CW_\TT)^+(\tilde z_\TS-\tilde z_\TT)\big\|^2\le\frac{\E\|\tilde z_\TS-\tilde z_\TT\|^2}{\sigma_{\min}(CW_\TT)^2}
=\frac{2V}{\sigma_{\min}(CW_\TT)^2}\lim_{\tau\to\infty}\E\cL_\tau .
\]
Final-layer feature matching therefore adds nothing to full-logit matching at high temperature. It is
informative only when outputs pass through a lossy channel: hard labels, top-$k$ logits, or $\tau=1$ with
a finite optimisation budget (\cref{prop:invisible}).
\end{proposition}
\begin{proof}
Take $A=(CW_\TT)^+CW_\TS$. With read-out biases, $\tilde z=C(Wh+b)$, and $(CW_\TT)^+CW_\TT=I$ gives
$Ah_\TS-h_\TT=Y-c$ with $Y:=(CW_\TT)^+(\tilde z_\TS-\tilde z_\TT)$ and the constant $c:=(CW_\TT)^+C(b_\TS-b_\TT)$. The features
are centred, so $\E[Ah_\TS-h_\TT]=0$, hence $c=\E Y$ and $\E\|Ah_\TS-h_\TT\|^2=\E\|Y-\E Y\|^2\le\E\|Y\|^2$. The last equality in
the statement is \cref{prop:temperature}(ii).
\end{proof}

\begin{proposition}[No identifiability gain in the realisable noiseless linear model]\label{prop:noidgain}
\Mo Let $u(s)\in\R^D$ be fixed input features, with teacher $h_\TT=B^\circ u\in\R^k$, $z_\TT=W^\circ h_\TT$, and student
$h_\TS=Bu\in\R^k$, $z_\TS=WBu$ (realisable, $d_\TS=d_\TT=k$). With exact soft labels on states whose features span a subspace
$\mathcal U\subseteq\R^D$, output matching identifies $CW^\circ B^\circ$ on $\mathcal U$. Adding exact representation
targets identifies $B^\circ$ on the same $\mathcal U$. Neither yields any extrapolation beyond $\mathcal U$, and
neither needs fewer states. Any gain must therefore come from noise, conditioning, or computation.
\end{proposition}

\subsection{When it provably helps}

\begin{proposition}[Noisy output supervision: fewer parameters learned from noise]\label{prop:repr-count}
\Kn In the model of \cref{prop:noidgain} (so $d_\TT=k$) with $\mathcal U=\R^D$, suppose outputs are observed only through
hard labels $y\sim\softmax(W^\circ B^\circ u)$ and $\rank(W^\circ B^\circ)=k$. The MLE over rank-$k$ factorisations
satisfies $n\,\E\KL\to\frac12k(V-1+D-k)$: the dimension of the manifold of rank-$k$ $(V-1)\times D$
matrices, by the regular-model asymptotics of \cref{prop:exact}(b). If $B^\circ$ is instead supplied exactly
by noiseless representation targets, only $W$ is estimated, and $n\,\E\KL\to\frac12k(V-1)$. The relative
reduction is $(D-k)/(V-1+D-k)$.
\end{proposition}
\noindent
In general, matching at a layer turns estimation of everything \emph{below} that layer into noiseless
regression. The statistical gain is the complexity of the sub-network below the matched layer relative
to the whole. The gain disappears when the output channel is itself noiseless (\cref{prop:noidgain}).

\begin{proposition}[Representation matching is a preconditioner]\label{prop:precond}
\Mo In the model of \cref{prop:noidgain}, near the optimum, with $W=W^\circ$ fixed and the near-lossless
quadratic output loss,
$\cL_{\mathrm{out}}(B)=\frac12\tr\big((B-B^\circ)\Sigma_u(B-B^\circ)^\top K\big)$ with $K=W^{\circ\top}FW^\circ$. Its Hessian is
$\Sigma_u\otimes K$. Adding $\gamma\E\|Bu-B^\circ u\|^2$ gives $\Sigma_u\otimes(K+2\gamma I)$, with the same minimiser.
Gradient descent converges linearly at rate $1-1/\kappa$, where
\[
\kappa_\gamma=\kappa(\Sigma_u)\,\frac{\lambda_{\max}(K)+2\gamma}{\lambda_{\min}(K)+2\gamma}\ \searrow\ \kappa(\Sigma_u)\quad(\gamma\to\infty).
\]
The ratio $\kappa_0/\kappa_\gamma$ can be arbitrarily large, and $\kappa_0=\infty$ when $K$ is singular, without any change
to the realisable solution. \emph{Heuristic:} in language models $K$ is badly conditioned, because
$F=\diag p-pp^\top$ is dominated by a few tokens (\cref{prop:invisible}), so the speed-up can be large in
practice.
\end{proposition}

\begin{proposition}[Compositional hardness: a computational separation]\label{prop:parity}
\Kn There are targets, for example $k$-sparse parities over $D$ bits computed through intermediate partial
parities, for which every statistical-query learner using only end-to-end labels needs $D^{\Omega(k)}$
queries. This follows from the statistical-query dimension of parities
\citep{blum1994weakly,kearns1998efficient}. Supervising the intermediate partial parities makes each step learnable
in polynomial time \citep{wies2023subtask}. The separation is \emph{computational}. Information-theoretically,
$\log\binom Dk\approx k\log D$ samples suffice end-to-end (\cref{thm:mle}). Representation matching can
therefore reduce the optimisation term $\cE_{\mathrm{opt}}$ exponentially, even when it cannot reduce $\cE_{\mathrm{est}}$.
\end{proposition}

\subsection{When it over-constrains, and a Fisher-aligned fix}

\begin{theorem}[Over-constraint in a linear-Gaussian model]\label{thm:overconstraint}
\Mo\Ck Let $h_\TT\sim\cN(0,\Sigma)$ in $\R^{d_\TT}$, with teacher centred logits $Wh_\TT$ and a constant Fisher matrix $F$.
Let the student be $h_\TS=Bh_\TT$ with $B\in\R^{d_\TS\times d_\TT}$ and $d_\TS<\rank\Sigma$, and let
$\cL_{\mathrm{out}}=\frac12\E\|F^{1/2}(W_\TS B-W)h_\TT\|^2$. Let $G:=F^{1/2}W\Sigma^{1/2}$ have singular values
$s_1\ge s_2\ge\cdots$, and let $(\lambda_i,u_i)$ be the eigenpairs of $\Sigma$.
\begin{enumerate}[label=(\alph*),nosep]
\item Output only ($\gamma=0$): $\min\cL_{\mathrm{out}}=\frac12\sum_{i>d_\TS}s_i^2$.
\item Full feature matching ($\gamma\to\infty$ with $\cR_{\mathrm{LS}}$): the row space of $B$ is forced onto
$\operatorname{span}(u_1,\dots,u_{d_\TS})$, provided $\lambda_{d_\TS}>\lambda_{d_\TS+1}$. Then
$\min\cL_{\mathrm{out}}\to\frac12\sum_{i>d_\TS}\lambda_i\|F^{1/2}Wu_i\|^2$.
\item The cost (b)$-$(a) is $\ge0$. It is $0$ iff the top-$d_\TS$ principal subspace of $\Sigma$ is output-optimal,
and it can be arbitrarily large. For example, with $d_\TT=2$, $d_\TS=1$, $\Sigma=\diag(\lambda_1,\lambda_2)$,
$\lambda_1>\lambda_2$, and $W=we_2^\top$: (a) is $0$ and (b) is $\frac12\lambda_2\,w^\top Fw$. The student is forced to
spend its only dimension on a high-variance nuisance direction.
\end{enumerate}
\end{theorem}
\begin{proof}
(a) $\cL_{\mathrm{out}}=\frac12\|F^{1/2}W_\TS B\Sigma^{1/2}-G\|_F^2$, and the first matrix has rank $\le d_\TS$; apply
Eckart--Young. (b) $\min_A\E\|ABh_\TT-h_\TT\|^2\ge\sum_{i>d_\TS}\lambda_i$, with equality iff $B=MU_{d_\TS}^\top$ for
invertible $M$ (PCA, when $\lambda_{d_\TS}>\lambda_{d_\TS+1}$). Given this $B$, the best read-out reproduces $W$ on the
retained subspace, leaving $\frac12\E\|F^{1/2}W(I-U_{d_\TS}U_{d_\TS}^\top)h_\TT\|^2$. (c) (a) is the unconstrained
minimum.
\end{proof}

\begin{definition}[Fisher-aligned representation matching]\label{def:fisher-rep}
Let $J(s)=\partial\tilde z_\TT(s)/\partial h_\TT(s)$ be the teacher's centred-logit Jacobian at the matched
layer; at the last layer $J=CW_\TT$. Let $K:=\E_\nu[J^\top F J]$ be the output-relevance matrix and $\Sigma$ the
covariance of $h_\TT$. Let $U_F$ collect the top-$d_\TS$ eigenvectors of $\Sigma^{1/2}K\Sigma^{1/2}$. Match the
\emph{output-relevant projection} of the teacher:
\[
\cR_{\mathrm F}(\theta):=\min_A\E_\nu\big\|Ah_\TS(s)-U_F^\top\Sigma^{-1/2}h_\TT(s)\big\|^2 .
\]
\end{definition}

\begin{proposition}[Fisher alignment has no approximation cost]\label{prop:fisher-rep}
\Mo\Ck In the model of \cref{thm:overconstraint}, assume $s_{d_\TS}>s_{d_\TS+1}$. For every $\gamma\ge0$, the minimisers
of $\cL_{\mathrm{out}}+\gamma\cR_{\mathrm F}$ are exactly the output-optimal students of case (a). The constraint costs
nothing in approximation. \emph{Heuristic:} it should retain the preconditioning effect of
\cref{prop:precond} in the retained directions.
\end{proposition}
\begin{proof}
Here $\Sigma^{1/2}K\Sigma^{1/2}=G^\top G$, so $U_F$ are the top right-singular vectors of $G$. The Eckart--Young
optimum is $F^{1/2}W_\TS B\Sigma^{1/2}=GU_FU_F^\top$. It is attained by $B=MU_F^\top\Sigma^{-1/2}$ and
$W_\TS=W\Sigma^{1/2}U_FM^{-1}$, for which $h_\TS=M\,U_F^\top\Sigma^{-1/2}h_\TT$ and so $\cR_{\mathrm F}=0$. Both terms are at
their minima simultaneously, so every minimiser of the sum has $\cR_{\mathrm F}=0$ and optimal $\cL_{\mathrm{out}}$.
The spectral gap gives uniqueness up to $M$.
\end{proof}

\begin{heuristic}[Deep networks]\label{heur:repr-deep}
For intermediate layers $J(s)$ varies with $s$, and $K$ averages over it. Alignment is then approximate,
but $K$ can be estimated with a few Jacobian--vector products per batch (Hutchinson-type probes). The
theory predicts the following. Full-feature matching helps when the teacher's high-variance directions
are output-relevant. It hurts when they carry nuisance variance, for example the outlier or ``massive''
activation dimensions reported in large transformers (we are unsure of the canonical reference for this
phenomenon). CKA matching over-constrains more than $\cR_{\mathrm{LS}}$ when $d_\TS<d_\TT$. Fisher-aligned matching
should dominate both.
\end{heuristic}
