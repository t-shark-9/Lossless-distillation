\section{Decoding-level losslessness}\label{sec:decoding}
% ===========================================================================

\Cref{prop:margin} is a pathwise, deterministic sufficient condition. This section answers the
quantitative question it leaves open: under the deployment law, how likely is it that the greedy or
beam-search outputs of teacher and student differ, given per-token divergences and the teacher's
margins? There are four answers.
(a)~Every $f$-divergence has an exact \emph{flip threshold}, which depends only on the teacher's two
largest probabilities (\cref{prop:flip}).
(b)~Given per-token errors along the teacher's greedy paths, the largest possible disagreement
probability is the value of a knapsack problem over the distribution of these thresholds. The value has a
closed dual form and is attained (\cref{thm:greedy}).
(c)~Sequence-level divergences, and hence the per-token errors that training controls, see the same
thresholds multiplied by the teacher's probability of its own greedy prefix. This weight is exact and
can be exponentially small in $T$, so behavioural and decoding losslessness are logically independent
(\cref{thm:greedy-seq,cor:separation}).
(d)~Beam search obeys the same laws, with the probability gap at the beam's cut in place of the margin
(\cref{thm:beam}).

\paragraph{Setting.}
Ties are broken by a fixed total order on $\cV$: $\gr(r)$ is the first maximiser of $r\in\Delta(\cV)$ in that
order. Greedy decoding of the teacher emits $y_t=\gr(p_t(\cdot\mid x,y_{<t}))$, and $g_p(x)\in\cY$ is its
output; $g_{q_\theta}(x)$ is defined in the same way. The teacher's greedy path is $s^\star_1=x$,
$a^\star_t=\gr(p_t(\cdot\mid s^\star_t))$, $s^\star_{t+1}=(s^\star_t,a^\star_t)$. At $s^\star_t$ we write
$p^\star_{t,1}\ge p^\star_{t,2}$ for the two largest teacher probabilities, $\gamma_t=p^\star_{t,1}-p^\star_{t,2}$ for the margin
of \cref{prop:margin}, $m_t=p^\star_{t,1}+p^\star_{t,2}$ for the top-two mass, and
\[
\pi^\star_t(x):=\Pseq\big(y_{<t}=a^\star_{<t}\big)=\prod_{j<t}p^\star_{j,1}
\]
for the teacher's probability of its own greedy prefix. All of these depend on $x$. The student
\emph{flips} at step $t$ if $\gr(q_t(\cdot\mid s^\star_t))\neq a^\star_t$. The \emph{decoding risk} is
$\Ldec(\theta;\mu):=\P_{x\sim\mu}\big[g_{q_\theta}(x)\neq g_p(x)\big]$.

\begin{lemma}[First flip; zero temperature]\label{lem:firstflip}
\Pv
\begin{enumerate}[label=(\roman*),nosep]
\item $g_{q_\theta}(x)\neq g_p(x)$ if and only if the student flips at some step. The two greedy outputs agree up
to the first flip and differ there.
\item For $\tau>0$, let $P^{x,\tau}$ and $Q^{x,\tau}_\theta$ be the sequence laws obtained by replacing every
next-token distribution $r$ by $r^{1/\tau}/\sum_ar(a)^{1/\tau}$. If the maximisers along both greedy paths are
unique, then $\TV(P^{x,\tau},Q^{x,\tau}_\theta)\to\ind\{g_{q_\theta}(x)\neq g_p(x)\}$ as $\tau\to0$. The decoding risk
is therefore the zero-temperature limit of the behavioural risk.
\end{enumerate}
\end{lemma}
\begin{proof}
(i) By induction on $t$: until the first flip, the student's greedy path visits the teacher's greedy
states. (ii) Each tempered next-token distribution tends to the point mass at its unique maximiser. Hence
$P^{x,\tau}(g_p(x))\to1$ and $Q^{x,\tau}_\theta(g_{q_\theta}(x))\to1$, and the TV between two point masses is the
indicator that they differ.
\end{proof}

\subsection{Flip thresholds}

\begin{definition}[Flip threshold; merge cost]\label{def:flip}
For a divergence $\mathsf D$ on $\Delta(\cV)$ and $p\in\Delta(\cV)$ with $a^\star=\gr(p)$, the \emph{flip threshold} is
\[
\kappa_{\mathsf D}(p):=\inf\Big\{\mathsf D(p\|q):\ q(a^\star)\le\max_{b\ne a^\star}q(b)\Big\}.
\]
Every flip satisfies the constraint, whatever the tie-breaking rule. For convex $f$ with $f(1)=0$, let
$g_f(u,v):=vf(u/v)$ be its perspective, with the conventions of \cref{def:fdiv}. For $P_1\ge P_2\ge0$ with
$R:=1-P_1-P_2\ge0$, the \emph{merge cost} is
\[
k_f(P_1,P_2):=\min_{0\le c\le\frac12}\Big[g_f(P_1,c)+g_f(P_2,c)+g_f(R,1-2c)\Big].
\]
\end{definition}

\noindent
The merge cost is the least $f$-divergence at which two atoms of a law, of masses $P_1\ge P_2$, can be put in
the wrong order. The next lemma proves this, and every threshold in this section is an instance of it.

\begin{lemma}[Merge lemma]\label{lem:merge}
\Pv\Ck Let $P$ be a law on a countable set, and let $S_+$, $S_-$ be disjoint finite sets of atoms with
$P^+:=\min_{S_+}P\ge P^-:=\max_{S_-}P$. For every $f$-divergence,
\[
\inf\big\{D_f(P\|Q):\ Q(a)\le Q(b)\text{ for some }a\in S_+,\ b\in S_-\big\}=k_f(P^+,P^-).
\]
The infimum is attained by the law that gives mass $c^\star$, a minimiser in $k_f$, to an atom $a^+\in S_+$ of
mass $P^+$ and to an atom $b^-\in S_-$ of mass $P^-$, and multiplies $P$ by $(1-2c^\star)/(1-P^+-P^-)$
elsewhere. If $P^++P^-=1$, the bracket in $k_f$ is non-increasing in $c$, and we take $c^\star=\frac12$. If $k_f(P^+,P^-)<\infty$, then for every $\delta>0$ some $Q$ with $Q(a^+)<Q(b^-)$ has
$D_f(P\|Q)\le k_f(P^+,P^-)+\delta$.
\end{lemma}
\begin{proof}
Let $Q(a)\le Q(b)$ with $a\in S_+$, $b\in S_-$ and $D_f(P\|Q)<\infty$.
\emph{Exchange.} If $Q(a^+)>Q(a)$, exchange the $Q$-masses of $a$ and $a^+$. Then, if $Q(b^-)<Q(b)$,
exchange the $Q$-masses of $b$ and $b^-$. Afterwards $Q(a^+)\le Q(b^-)$. Neither exchange increases $D_f$.
Each one gives the larger of two $Q$-masses to the atom with the larger $P$-mass, and the perspective is
submodular: $g_f(u',v')+g_f(u,v)\le g_f(u',v)+g_f(u,v')$ for $u\le u'$ and $v\le v'$. To see this, note
that for $v>0$, $g_f(u',v)-g_f(u,v)$ is $u'-u$ times the slope of the chord of $f$ between $u/v$ and $u'/v$.
Both endpoints decrease as $v$ grows, and chord slopes of a convex function increase with either
endpoint. The case $v=0$ follows by continuity.
\emph{Three cells.} Data processing onto the partition $\{a^+\}$, $\{b^-\}$, rest gives
$D_f(P\|Q)\ge D_f(\bar P\|\bar Q)$, with $\bar P=(P^+,P^-,R)$ and $\bar Q_1\le\bar Q_2$.
\emph{Equalise.} The map $\bar Q\mapsto D_f(\bar P\|\bar Q)$ is convex and vanishes at $\bar P$, which violates the
constraint if $P^+>P^-$ (if $P^+=P^-$, both sides are $0$). So the segment from $\bar Q$ to $\bar P$ meets
$\{\bar Q_1=\bar Q_2\}$ at a point of no larger divergence, and $D_f(P\|Q)\ge k_f(P^+,P^-)$. The stated law
has a likelihood ratio that is constant on each cell, so its divergence equals the three-cell value
$k_f(P^+,P^-)$. (If $R=0$, the derivative of the bracket is $\psi(P^+/c)+\psi(P^-/c)-2\psi(0)\le0$, where
$\psi(u)=f(u)-uf'(u)$ is non-increasing, so $c^\star=\frac12$ is a minimiser.)
\emph{Strict order.} The least $D_f(P\|Q)$ subject to $Q(b^-)\ge Q(a^+)+\delta$ is convex and non-decreasing
in $\delta$ and equals $k_f$ at $\delta=0$. It is finite for small $\delta>0$: move mass $\delta$ into $b^-$ from
another atom of the minimiser. Hence it tends to $k_f$ as $\delta\downarrow0$.
\end{proof}

\begin{proposition}[Exact flip thresholds]\label{prop:flip}
\Pv\Ck Let $p_1\ge p_2$ be the two largest probabilities of $p$, $\gamma=p_1-p_2$ and $m=p_1+p_2$.
\begin{enumerate}[label=(\roman*),nosep]
\item For every $f$-divergence, $\kappa_f(p)=k_f(p_1,p_2)$: only the two largest probabilities matter. The
infimum is attained, and for every $\delta>0$ a strict flip (one that flips under any tie rule) costs at
most $k_f(p_1,p_2)+\delta$.
\item Closed forms and lower bounds:
\begin{center}\small
\begin{tabular}{@{}lll@{}}
\toprule
$\mathsf D(p\|q)$ & $\kappa_{\mathsf D}(p)$ & lower bound\\\midrule
$\TV$ & $\gamma/2$ & --\\
$\KL(p\|q)$ & $p_1\log\frac{2p_1}{m}+p_2\log\frac{2p_2}{m}$ & $\gamma^2/(2m)$\\
$\KL(q\|p)$ & $-\log\big(1-(\sqrt{p_1}-\sqrt{p_2})^2\big)$ & $\gamma^2/(2m)$\\
$\Hel^2$ & $1-\sqrt{1-\frac12(\sqrt{p_1}-\sqrt{p_2})^2}$ & $\gamma^2/(8m)$\\
$\chi^2(p\|q)$ & $\big(1-m+\sqrt{m^2+\gamma^2}\big)^2-1$ & $2(\sqrt{m^2+\gamma^2}-m)$\\
$\JSD_\beta(p\|q)$ & $k_{f_\beta}(p_1,p_2)$ & $\beta(1-\beta)\gamma^2/2$\\
logit oscillation $\max_a\log\frac{q(a)}{p(a)}-\min_a\log\frac{q(a)}{p(a)}$ & $\log(p_1/p_2)$ & --\\
centred logits $\big\|C(\log q-\log p)\big\|_2$ & $\log(p_1/p_2)/\sqrt2$ & --\\\bottomrule
\end{tabular}
\end{center}
\item (Local form.) If $f$ is twice continuously differentiable near $1$ with $f''(1)>0$, then
$\kappa_f(p)=\frac{f''(1)\,\gamma^2}{2m}\,(1+o(1))$ as $\gamma\to0$ with $m$ fixed. For forward KL, reverse KL and
$\Hel^2$, this leading term is also a global lower bound.
\end{enumerate}
\end{proposition}
\begin{proof}
(i) Apply \cref{lem:merge} with $S_+=\{a^\star\}$ and $S_-=\cV\setminus\{a^\star\}$.
(ii) For TV the bracket in $k_f$ is $\frac12(|p_1-c|+|p_2-c|+|m-2c|)$, minimal at $c=m/2$. For forward KL it
is $p_1\log\frac{p_1}c+p_2\log\frac{p_2}c+R\log\frac R{1-2c}$, which is stationary at $c=m/2$. For reverse KL it is
$c\log\frac{c^2}{p_1p_2}+(1-2c)\log\frac{1-2c}R$. Stationarity gives $c=\sqrt{p_1p_2}/Z$ with
$Z=R+2\sqrt{p_1p_2}=1-(\sqrt{p_1}-\sqrt{p_2})^2$, and the value is $-\log Z$. For $\Hel^2$, maximise
$\sqrt{2c}\cdot\frac{\sqrt{p_1}+\sqrt{p_2}}{\sqrt2}+\sqrt{1-2c}\cdot\sqrt R$ by Cauchy--Schwarz. For $\chi^2$, minimise
$\frac{2(p_1^2+p_2^2)}{2c}+\frac{R^2}{1-2c}$ using $\frac{a^2}x+\frac{b^2}y\ge\frac{(a+b)^2}{x+y}$, and note
$2(p_1^2+p_2^2)=m^2+\gamma^2$. For the lower bounds: the KL threshold is $m\,\mathrm{kl}(\frac{p_1}m\|\frac12)\ge2m(\frac\gamma{2m})^2$
by Pinsker. Next, $-\log(1-x)\ge x$, $1-\sqrt{1-x/2}\ge x/4$, and
$(\sqrt{p_1}-\sqrt{p_2})^2=\gamma^2/(\sqrt{p_1}+\sqrt{p_2})^2\ge\gamma^2/(2m)$. The $\chi^2$ bound drops the square,
and the $\JSD_\beta$ bound combines \cref{prop:jsd}(i) with $\TV\ge\gamma/2$. For logits, let
$\delta=\log q-\log p$. A flip to $b$ needs $\delta_b-\delta_{a^\star}\ge\log\frac{p(a^\star)}{p(b)}\ge\log\frac{p_1}{p_2}$, and
$\|C\delta\|_2^2=\min_c\|\delta-c\one\|^2\ge\frac12(\delta_b-\delta_{a^\star})^2$. The perturbation
$\delta=\frac12\log\frac{p_1}{p_2}\,(e_{b}-e_{a^\star})$, with $b$ the runner-up, attains both.
(iii) The choice $c=m/2$ gives $\frac m2[f(1+\frac\gamma m)+f(1-\frac\gamma m)]=\frac{f''(1)\gamma^2}{2m}(1+o(1))$.
Conversely, write the bracket as $A+B$. Here $B:=2cf(\frac m{2c})+(1-2c)f(\frac R{1-2c})=D_f(\mathrm{Bern}(m)\|\mathrm{Bern}(2c))\ge0$,
and $A:=c\,[f(u_+)+f(u_-)-2f(\frac{u_++u_-}2)]$ with $u_\pm=p_{1,2}/c$. At a minimiser $A+B=O(\gamma^2)$, so
$B\to0$ forces $2c\to m$. Then $u_\pm\to1$ and $A=c\,(\frac{\gamma}{2c})^2f''(\xi)$ with $\xi\to1$. The global
lower bounds are those of (ii).
\end{proof}

\noindent
Near a tie, TV needs error $\gamma/2$ to flip, but every smooth divergence needs only order $\gamma^2/m$. The
factor $1/m$ is invisible to Pinsker's inequality, which gives only $\KL\ge\gamma^2/2$.

\subsection{Greedy decoding: errors on the greedy path}

\begin{definition}[Knapsack function]\label{def:knapsack}
For a cost $\omega:\cX\to[0,\infty]$ and a budget $\eps\ge0$,
\[
\Phi_\omega(\eps):=\sup\Big\{\mu(A):\ \int_A\omega\dd\mu\le\eps\Big\},\qquad
\bar\Phi_\omega(\eps):=\inf_{\tau>0}\Big\{\frac\eps\tau+\E_\mu\Big(1-\frac\omega\tau\Big)_+\Big\}.
\]
$\Phi_\omega(\eps)$ is the largest $\mu$-mass that a budget $\eps$ can buy when input $x$ costs $\omega(x)$.
\end{definition}

\begin{lemma}[Knapsack duality]\label{lem:knapsack}
\Pv\Ck $\bar\Phi_\omega(\eps)=\sup\{\E_\mu w:\ 0\le w\le1,\ \E_\mu[w\omega]\le\eps\}$. This fractional problem is
solved by buying inputs in increasing order of $\omega$. Moreover
$\bar\Phi_\omega-\max_x\mu(\{x\})\le\Phi_\omega\le\bar\Phi_\omega$ and
$\bar\Phi_\omega(\eps)\le\inf_{\tau>0}\{\eps/\tau+\mu[\omega<\tau]\}$.
\end{lemma}
\begin{proof}
For feasible $w$ and $\tau>0$,
$\E w\le\E w+\frac1\tau(\eps-\E[w\omega])=\frac\eps\tau+\E[w(1-\omega/\tau)]\le\frac\eps\tau+\E(1-\omega/\tau)_+$.
Now let $w$ buy inputs in increasing order of $\omega$ until the budget is spent, with a fractional purchase
only at the last level $\tau^\star$ reached. Then $w=1$ on $\{\omega<\tau^\star\}$, $w=0$ on $\{\omega>\tau^\star\}$ and
$\E[w\omega]=\eps$, so both inequalities are equalities at $\tau=\tau^\star$. If the budget is never exhausted,
they become equalities as $\tau\to\infty$. On a countable space the fractional purchase can be confined to a
single atom, and dropping that atom gives a feasible set. Finally,
$(1-\omega/\tau)_+\le\ind\{\omega<\tau\}$.
\end{proof}

\begin{theorem}[Greedy decoding: sharp bounds from errors on the greedy path]\label{thm:greedy}
\Pv\Ck Let $\mathsf D$ be a divergence on $\Delta(\cV)$. Put $e^\star_t(x):=\mathsf D(p_t\|q_t)(s^\star_t)$,
$\kappa^\star_t(x):=\kappa_{\mathsf D}(p_t(\cdot\mid s^\star_t))$ and $\bar\kappa(x):=\min_t\kappa^\star_t(x)$.
\begin{enumerate}[label=(\roman*),nosep]
\item (Pathwise.) $\ind\{g_{q_\theta}(x)\ne g_p(x)\}\le\max_t\ind\{e^\star_t\ge\kappa^\star_t\}\le\min\{1,\max_te^\star_t/\kappa^\star_t\}$.
\item (Probability.) Let $\eps^\star:=\E_\mu\sum_te^\star_t$ be the summed per-token error on greedy paths. Then
\begin{gather*}
\Ldec\le\Phi_{\bar\kappa}\Big(\E_\mu\max_te^\star_t\Big)\le\bar\Phi_{\bar\kappa}(\eps^\star)\le\inf_{\tau>0}\Big\{\mu[\bar\kappa<\tau]+\frac{\eps^\star}\tau\Big\},\\
\Ldec\le\sum_{t=1}^T\bar\Phi_{\kappa^\star_t}\big(\E_\mu e^\star_t\big).
\end{gather*}
\item (Sharpness.) If $\mathsf D$ is an $f$-divergence then, for every teacher, input law and $\eps>0$,
\[
\bar\Phi_{\bar\kappa}(\eps)-\max_x\mu(\{x\})\ \le\ \sup_{q_\theta}\Big\{\Ldec(\theta;\mu):\ \E_\mu\textstyle\sum_te^\star_t\le\eps\Big\}\ \le\ \Phi_{\bar\kappa}(\eps),
\]
with equality on the right wherever $\Phi_{\bar\kappa}$ is left-continuous. The same holds with $\max_t$ in place of
$\sum_t$. The extremal students differ from the teacher at one greedy state per input.
\item (Uniform errors.) If $e^\star_t(x)\le\eta$ for all $t$ and $x$, then $\Ldec\le\mu[\bar\kappa\le\eta]$. For $f$-divergences,
some such student reaches $\mu[\bar\kappa<\eta]$.
\end{enumerate}
\end{theorem}
\begin{proof}
(i) By \cref{lem:firstflip}, a disagreement needs a flip at some $s^\star_t$, and a flip needs
$e^\star_t\ge\kappa^\star_t$ by the definition of $\kappa_{\mathsf D}$.
(ii) Let $A$ be the disagreement set. By (i), $\bar\kappa\le\max_te^\star_t$ on $A$, so
$\int_A\bar\kappa\dd\mu\le\E_\mu\max_te^\star_t$. Then use $\max\le\sum$ and \cref{lem:knapsack}. For the second line,
apply the same argument to each flip event and take a union bound.
(iii) The right inequality is (ii). For the left, take $A$ with $\int_A\bar\kappa\dd\mu<\eps$. For $x\in A$, let
$t(x)$ minimise $\kappa^\star_t(x)$, and let the student equal the teacher except at the states
$s^\star_{t(x)}(x)$, $x\in A$. There it uses a strict flip of cost at most $\bar\kappa(x)+\delta$
(\cref{prop:flip}(i)). States of different inputs are different, so this defines a student. It disagrees
with the teacher exactly on $A$, and $\E\sum_te^\star_t=\E\max_te^\star_t\le\int_A\bar\kappa\dd\mu+\delta<\eps$ for small
$\delta$. So the supremum is at least $\sup\{\mu(A):\int_A\bar\kappa<\eps\}=\lim_{\eps'\uparrow\eps}\Phi_{\bar\kappa}(\eps')$.
This is at least $\bar\Phi_{\bar\kappa}(\eps)-\max_x\mu(\{x\})$ by \cref{lem:knapsack}, because $\bar\Phi_{\bar\kappa}$ is
concave and hence continuous on $(0,\infty)$.
(iv) By (i), a disagreement needs $\bar\kappa\le e^\star_t\le\eta$. For the lower bound, flip every input with
$\bar\kappa<\eta$ as in (iii).
\end{proof}

\begin{remark}[Reading \cref{thm:greedy}]\label{rem:greedy}
Greedy decoding is plug-in classification of the teacher's argmax at each greedy state
\citep{audibert2007fast}. The knapsack
function plays the role that the calibration function of a surrogate loss plays in classification
\citep{bartlett2006convexity}. Here it is exact, because the teacher is fixed. The worst student spends
its error budget on the inputs whose greedy paths contain the cheapest flips, in increasing order of
$\bar\kappa$. There are two regimes. If $\bar\kappa\ge\kappa_0>0$ on the support of $\mu$, then $\Ldec\le\eps^\star/\kappa_0$,
and $\Ldec=0$ as soon as every $e^\star_t<\kappa_0$. If instead $\bar\kappa$ has mass near $0$, a budget $\eps$ buys a
disagreement probability much larger than $\eps$ (\cref{cor:rates}). The per-step form in (ii) uses the
per-position error curves of \cref{sec:experiments}, evaluated on greedy prefixes.
\end{remark}
